# Eclipse Survivors — ระบบแมพ: สิ่งกีดขวาง เนิน แอ่ง จุดสำคัญ ประตูมิติ อีเวนต์

> **วิธีใช้:** ส่งให้ Gemini ทีละ Task ตามลำดับ ลองเล่นหลังจบแต่ละ Task โดยเฉพาะเรื่อง **ความลื่นของเกม (FPS)** บนมือถือ
> **หลักสำคัญ:** ทุกอย่างบนแมพสร้างจาก **hash ของช่องตาราง** (แบบเดียวกับ `getVisible25DProps` ใน `renderer.ts`) ตำแหน่งจึงคงที่ ไม่ต้องเก็บลงหน่วยความจำ และใช้ได้กับแมพไม่มีขอบ

## ส่วนที่ 0: สภาพแมพตอนนี้ (จากโค้ด)

- พื้นเป็นช่องข้าวหลามตัด isometric (`ISO_TILE_HW = 92`, `ISO_TILT = 0.68`)
- `getVisible25DProps` สุ่มของตกแต่งจาก hash ของช่อง (`hash > 0.908`) และเว้นรอบจุดเกิด 170px
- ของตกแต่ง **เป็นแค่ภาพ** ไม่มีการชน มีแค่โปร่งใสตอนผู้เล่นเดินไปอยู่ข้างหลัง
- พิกัดโลก: `wx = (ix - iy) × 92`, `wy = (ix + iy) × 92` และแปลงกลับได้ด้วย `ix = (wx + wy) / 184`, `iy = (wy - wx) / 184`

## Task M1: สิ่งกีดขวางทรงกลม (ทำก่อน สำคัญที่สุด)

### M1.1 แนวคิด
- ของตกแต่งบางส่วนกลายเป็น **สิ่งกีดขวาง** ที่มีวงชน (หิน เสา ต้นไม้ใหญ่ ซากกำแพง)
- ผู้เล่นและมอน **ชนแล้วไถลอ้อม** ไม่ต้องหาเส้นทาง (ไม่มี pathfinding)
- **กระสุนของมอนชนแล้วหายไป** → ผู้เล่นหลบหลังเสาได้ ช่วยให้เกมมีมิติขึ้นมาก
- กระสุนของผู้เล่น **ทะลุได้** (กันผู้เล่นรู้สึกว่าอาวุธยิงไม่โดน) ยกเว้นอาวุธหนักที่ระบุในหัวข้ออื่น
- บอสและยมทูต **เดินทะลุ** สิ่งกีดขวาง (กันติดค้าง) และทำลายสิ่งที่ตีแตกได้ที่เดินผ่าน

### M1.2 ไฟล์ใหม่ `src/game/mapObstacles.ts`
```ts
export const MAP_TILE_HW = 92;

export interface MapObstacle {
  id: string;
  x: number; // พิกัดโลก (ก่อนเอียง)
  y: number;
  r: number; // รัศมีชน
  kind: 'rock' | 'pillar' | 'bigTree' | 'ruin';
  variant: number;
  scale: number;
}

// hash เดียวกับของตกแต่ง แต่ใช้ seed ต่างกัน เพื่อไม่ให้ทับกัน
function tileHash(ix: number, iy: number, seed: number): number {
  return Math.abs(Math.sin(ix * 12.9898 + iy * 78.233 + seed * 37.719) * 43758.5453) % 1;
}

// ความหนาแน่น: 0.06 = ประมาณ 6% ของช่องมีสิ่งกีดขวาง (ปรับได้)
export const OBSTACLE_DENSITY = 0.06;
export const SPAWN_CLEAR_RADIUS = 260;

export function getObstacleAtTile(ix: number, iy: number, runSeed: number): MapObstacle | null {
  const h = tileHash(ix, iy, runSeed);
  if (h > OBSTACLE_DENSITY) return null;
  const x = (ix - iy) * MAP_TILE_HW;
  const y = (ix + iy) * MAP_TILE_HW;
  if (Math.hypot(x, y) < SPAWN_CLEAR_RADIUS) return null;
  const k = Math.floor((h / OBSTACLE_DENSITY) * 4);
  const kind = (['rock', 'pillar', 'bigTree', 'ruin'] as const)[Math.min(3, k)];
  const scale = 0.9 + ((h * 1000) % 1) * 0.5;
  const baseR = kind === 'pillar' ? 22 : kind === 'bigTree' ? 30 : kind === 'ruin' ? 40 : 34;
  return { id: `${ix}:${iy}`, x, y, r: baseR * scale, kind, variant: Math.floor((h * 7919) % 4), scale };
}

// สิ่งกีดขวางรอบจุด (x,y) ในรัศมี range — ใช้ทั้งการชนและการวาด
export function getObstaclesNear(x: number, y: number, range: number, runSeed: number): MapObstacle[] {
  const out: MapObstacle[] = [];
  const cIx = (x + y) / (2 * MAP_TILE_HW);
  const cIy = (y - x) / (2 * MAP_TILE_HW);
  const span = Math.ceil(range / MAP_TILE_HW) + 1;
  for (let ix = Math.floor(cIx) - span; ix <= Math.ceil(cIx) + span; ix++) {
    for (let iy = Math.floor(cIy) - span; iy <= Math.ceil(cIy) + span; iy++) {
      const o = getObstacleAtTile(ix, iy, runSeed);
      if (o && Math.hypot(o.x - x, o.y - y) < range + o.r) out.push(o);
    }
  }
  return out;
}

// ดันวงกลม (ผู้เล่น/มอน) ออกจากสิ่งกีดขวาง และเพิ่มแรงไถลด้านข้างกันติด
export function resolveCircleVsObstacles(
  ent: { x: number; y: number },
  entR: number,
  obstacles: MapObstacle[],
  slideBias = 0.35
): boolean {
  let hit = false;
  for (const o of obstacles) {
    const dx = ent.x - o.x;
    const dy = ent.y - o.y;
    const d = Math.hypot(dx, dy) || 0.0001;
    const minD = o.r + entR;
    if (d < minD) {
      const nx = dx / d;
      const ny = dy / d;
      const push = minD - d;
      ent.x += nx * push;
      ent.y += ny * push;
      // ไถลไปทางข้าง (ตั้งฉากกับแนวชน) เพื่อไม่ให้มอนติดหน้าหิน
      ent.x += -ny * push * slideBias;
      ent.y += nx * push * slideBias;
      hit = true;
    }
  }
  return hit;
}
```

### M1.3 ต่อเข้ากับเกม (`GameCanvas.tsx`)
1. ตอนเริ่มรอบ สุ่ม `sim.mapSeed = Math.floor(Math.random() * 100000)` (แมพต่างกันทุกรอบ แต่คงที่ภายในรอบ)
2. **แคชต่อเฟรม:** ต้นเฟรม เรียก `getObstaclesNear(player.x, player.y, 900, sim.mapSeed)` ครั้งเดียว เก็บใน `frameObstacles`
3. **ผู้เล่น:** หลังขยับผู้เล่น (ทั้งเดินปกติ `sim.player.x += sim.player.vx * dt` และแดช `sim.dashVx`) เรียก `resolveCircleVsObstacles(sim.player, 14, frameObstacles)`
   - ระหว่างแดช/ฝูงค้างคาว **ไม่ต้องชน** (ผู้เล่นทะลุได้ ให้รางวัลการใช้แดช)
4. **มอน:** หลังบล็อกการเคลื่อนที่ของมอนแต่ละตัว ถ้ามอนอยู่ในระยะ 900px จากผู้เล่น ให้เรียก `resolveCircleVsObstacles(e, e.radius, frameObstacles)` (มอนนอกระยะไม่ต้องเช็ก ประหยัดเครื่อง)
   - ยกเว้น `e.isBoss` (เดินทะลุ)
   - มอนบิน (ค้างคาว ผี) **บินข้าม** ได้ ไม่ต้องชน
5. **กระสุนมอน:** ทุกนัด ถ้าอยู่ใน `frameObstacles` วงใดวงหนึ่ง → ลบกระสุน + อนุภาคฝุ่นเล็กๆ
6. **วาด:** ใส่สิ่งกีดขวางเข้าไปในรายการวาดที่เรียงตาม y เดียวกับ `draw25DProp` (มีอยู่แล้วตรง `visibleProps`) และทำให้โปร่งใสตอนผู้เล่นอยู่ข้างหลังเหมือนของตกแต่ง
   - วาด **เงาวงรีบนพื้น** ขนาดเท่าวงชน ผู้เล่นจะเห็นชัดว่าชนตรงไหน
7. **ของตกแต่งเดิม:** ถ้าช่องไหนมีสิ่งกีดขวาง ไม่ต้องวาดของตกแต่งทับ
